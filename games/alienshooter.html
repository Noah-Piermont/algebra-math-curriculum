---------------------------------------------------------------------------- # State transitions # ---------------------------------------------------------------------------- def set_cursor(on): cur = getattr(player, 'cursor', None) if player else None if cur is not None: cur.enabled = on def start_game(): global state, player, player_hp, shoot_cd, spawn_timer, play_start, game_time global session_kills, session_gold clear_enemies() build_map(save['map']) if player is None: player = FirstPersonController() player.speed = 6.5 player.enabled = True set_cursor(True) player.position = Vec3(0, 1, 0) player.rotation_y = 0 player.camera_pivot.rotation_x = 0 mouse.locked = True player_hp = PLAYER_MAX_HP shoot_cd = 0.0 spawn_timer = 1.5 session_kills = 0 session_gold = 0 play_start = game_time apply_gun() apply_skin() viewmodel.enabled = True hud.enabled = True refresh_hud() show_panel(None) state = 'playing' def game_over(): global state state = 'dead' write_save() player.enabled = False set_cursor(False) viewmodel.enabled = False mouse.locked = False mouse.visible = True over_stats.text = f'Kills: {session_kills} Gold earned: {session_gold} Total gold: {save["gold"]}' show_panel('over') def go_menu(): global state state = 'menu' clear_enemies() clear_world() if player is not None: player.enabled = False set_cursor(False) viewmodel.enabled = False hud.enabled = False window.color = MENU_BG mouse.locked = False mouse.visible = True write_save() refresh_menu() show_panel('main') # ---------------------------------------------------------------------------- # Main loop hooks # ---------------------------------------------------------------------------- def update(): global game_time, shoot_cd, spawn_timer if state != 'playing': return dt = time.dt game_time += dt shoot_cd = max(0.0, shoot_cd - dt) if held_keys['left mouse'] and game_time - play_start > .3: try_shoot() spawn_timer -= dt if spawn_timer <= 0: max_alive = min(10, 3 + session_kills // 4) if len(aliens) < max_alive: spawn_alien() spawn_timer = max(1.4, 4.0 - session_kills * .08) if player.y < -15: player.position = Vec3(0, 3, 0) def input(key): if state == 'playing' and key == 'escape': go_menu() # ---------------------------------------------------------------------------- # Go! # ---------------------------------------------------------------------------- apply_gun() apply_skin() refresh_menu() show_panel('main') mouse.locked = False mouse.visible = True app.run()
